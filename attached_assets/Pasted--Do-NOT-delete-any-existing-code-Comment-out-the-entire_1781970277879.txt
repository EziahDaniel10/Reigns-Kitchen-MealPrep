jsimport express from 'express';
import cors from 'cors';
import dotenv from 'dotenv';
import fetch from 'node-fetch';
import { Client } from "@replit/database";

dotenv.config();
const app = express();
app.use(cors());
app.use(express.json());
const db = new Client();

function getRecipients() {
  const recipients = [];
  let i = 1;
  while (process.env[`CALLMEBOT_PHONE_${i}`]) {
    recipients.push({
      phone: process.env[`CALLMEBOT_PHONE_${i}`],
      apikey: process.env[`CALLMEBOT_APIKEY_${i}`]
    });
    i++;
  }
  return recipients;
}

async function sendWhatsApp(phone, apikey, message) {
  const encoded = encodeURIComponent(message);
  const url = `https://api.callmebot.com/whatsapp.php?phone=${phone}&text=${encoded}&apikey=${apikey}`;
  try {
    const response = await fetch(url);
    return response.ok;
  } catch {
    return false;
  }
}

function cleanPrice(value) {
  return Number(String(value).replace(/[^0-9.]/g, ''));
}

function formatOrderMessage(order) {
  const { customerName, customerPhone, customerEmail, pickupTime, items, note, orderNumber } = order;

  const itemList = items.map(item => {
    const modifiers = [];
    if (item.combo) modifiers.push(`Combo (${item.comboDrink})`);
    if (item.hushPuppies) modifiers.push('+ Hush Puppies');
    if (item.sideUpgrade) modifiers.push(item.sideUpgrade);
    if (item.sauce) modifiers.push(`Sauce: ${item.sauce}`);

    const modText = modifiers.length ? `\n   ${modifiers.join(' | ')}` : '';
    const price = cleanPrice(item.price);
    const qty = parseInt(item.qty, 10);
    const lineTotal = (price * qty).toFixed(2);

    return `• ${item.name} x${qty} — $${lineTotal}${modText}`;
  }).join('\n');

  const grandTotal = items.reduce((sum, item) => {
    return sum + cleanPrice(item.price) * parseInt(item.qty, 10);
  }, 0).toFixed(2);

  return {
    text: `
🦐 NEW ORDER — Reigns Kitchen
━━━━━━━━━━━━━━━━━━
🔖 Order #${orderNumber}
👤 Name: ${customerName}
📞 Phone: ${customerPhone}
📧 Email: ${customerEmail}
🕐 Pickup Time: ${pickupTime}
━━━━━━━━━━━━━━━━━━
ORDER:
${itemList}
━━━━━━━━━━━━━━━━━━
💰 TOTAL: $${grandTotal}
${note ? `📝 Note: ${note}` : ''}
━━━━━━━━━━━━━━━━━━
    `.trim(),
    grandTotal
  };
}

function generateOrderNumber() {
  const date = new Date();
  const dateStr = date.toISOString().slice(0,10).replace(/-/g,'');
  const rand = Math.floor(Math.random() * 1000).toString().padStart(3, '0');
  return `RK-${dateStr}-${rand}`;
}

app.post('/api/send-order', async (req, res) => {
  try {
    const orderNumber = generateOrderNumber();
    const order = { ...req.body, orderNumber, timestamp: new Date().toISOString() };

    // STEP 1 — Save order FIRST, before any notification attempt
    // This guarantees the order is never lost even if WhatsApp/email fail
    await db.set(`order:${orderNumber}`, order);

    const { text: message, grandTotal } = formatOrderMessage(order);

    // STEP 2 — WhatsApp via CallMeBot (both numbers)
    const recipients = getRecipients();
    const whatsappResults = await Promise.all(
      recipients.map(r => sendWhatsApp(r.phone, r.apikey, message))
    );
    const whatsappDelivered = whatsappResults.some(r => r === true);

    // Update saved order with delivery status
    order.whatsappDelivered = whatsappDelivered;
    order.total = grandTotal;
    await db.set(`order:${orderNumber}`, order);

    res.json({
      success: true,
      orderNumber,
      total: grandTotal,
      whatsappDelivered
    });

  } catch (err) {
    console.error('Order error:', err);
    res.status(500).json({ success: false, error: 'Failed to process order. Please try again.' });
  }
});

// Backup safety-net view — only used if WhatsApp + email both fail
app.get('/api/admin/orders', async (req, res) => {
  const { password } = req.query;
  if (password !== process.env.ADMIN_PASSWORD) {
    return res.status(401).json({ error: 'Unauthorized' });
  }
  const keys = await db.list('order:');
  const orders = await Promise.all(keys.map(k => db.get(k)));
  orders.sort((a, b) => new Date(b.timestamp) - new Date(a.timestamp));
  res.json(orders);
});

app.get('/api/config', (req, res) => {

async function submitOrder() {
  const order = {
    customerName: document.getElementById('customer-name').value.trim(),
    customerEmail: document.getElementById('customer-email').value.trim(),
    customerPhone: document.getElementById('customer-phone').value.trim(),
    pickupTime: document.getElementById('pickup-time').value,
    note: document.getElementById('order-note').value.trim(),
    items: cart.map(item => ({
      name: item.name,
      qty: parseInt(item.qty, 10),
      price: Number(String(item.price).replace(/[^0-9.]/g, '')),
      combo: item.combo || false,
      comboDrink: item.comboDrink || '',
      hushPuppies: item.hushPuppies || false,
      sideUpgrade: item.sideUpgrade || '',
      sauce: item.sauce || ''
    }))
  };

  if (!order.customerName || !order.customerEmail || !order.customerPhone || !order.pickupTime) {
    showError('Please fill in all required fields.');
    return;
  }
  if (order.items.length === 0) {
    showError('Your order is empty.');
    return;
  }

  const btn = document.getElementById('submit-btn');
  btn.disabled = true;
  btn.textContent = 'Placing order...';

  try {
    const res = await fetch('/api/send-order', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(order)
    });
    const data = await res.json();

    if (!data.success) {
      showError(data.error || 'Something went wrong. Please try again.');
      return;
    }

    const itemsFormatted = order.items
      .map(i => `${i.name} x${i.qty} — $${(i.price * i.qty).toFixed(2)}`)
      .join('\n');
    const noteText = order.note ? `Note: ${order.note}` : '';

    // Customer confirmation email
    await emailjs.send(emailConfig.emailjsServiceId, emailConfig.emailjsCustomerTemplateId, {
      customer_name: order.customerName,
      customer_email: order.customerEmail,
      order_number: data.orderNumber,
      pickup_time: order.pickupTime,
      order_items: itemsFormatted,
      order_total: data.total,
      customer_note: noteText
    });

    // Owner backup email (sends to fixed address set inside the EmailJS template)
    await emailjs.send(emailConfig.emailjsServiceId, emailConfig.emailjsOwnerTemplateId, {
      customer_name: order.customerName,
      customer_email: order.customerEmail,
      customer_phone: order.customerPhone,
      order_number: data.orderNumber,
      pickup_time: order.pickupTime,
      order_items: itemsFormatted,
      order_total: data.total,
      customer_note: noteText
    });

    showConfirmation(data.orderNumber, order.customerEmail);

  } catch (err) {
    console.error(err);
    showError('Connection error. Please try again or contact us directly.');
  } finally {
    btn.disabled = false;
    btn.textContent = 'Place Order';
  }
}

      <p>See you at pickup!</p>
      <button onclick="closeModal()">Back to Menu</button>
    </div>
  `;
}

✅ Summary of Notification Flow

Order saved permanently the instant it's submitted (never lost)
WhatsApp sent to both numbers via CallMeBot (primary)
Email sent to customer (confirmation)
Email sent to client (backup, in case WhatsApp silently fails)
/api/admin/orders?password=... exists as a last-resort safety net, not meant for routine checking
